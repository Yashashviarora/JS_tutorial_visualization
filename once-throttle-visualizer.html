🔂 once() & 🚰 throttle()

Two more closure utilities, from the same family as the counter factory and debounce. Each one is a higher-order function: it takes your function and returns a guarded version of it, with one remembered variable in its 🎒 backpack deciding whether your function is allowed to run.

1once(): the function can only run one time

function once(fn) {
  let called = false;          // 🎒 "have I run already?"
  let result;                  // 🎒 the result of the first run
  return function (...args) {
    if (called) return result;  // every later call: skip fn, return the saved result
    called = true;
    result = fn.apply(this, args);
    return result;
  };
}
const payOnce = once(() => chargeCard(499));

2throttle(): at most once every N milliseconds

function throttle(fn, limit) {
  let lastRun = 0;                          // 🎒 when did fn last run?
  return function (...args) {
    const now = Date.now();
    if (now - lastRun < limit) return;      // too soon since the last run: ignore this call
    lastRun = now;
    fn.apply(this, args);
  };
}
